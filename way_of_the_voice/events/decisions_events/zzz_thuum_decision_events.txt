				compare_modifier = {
					value = scope:duel_value

					min = -19
				}

                add_stress = -75
                add_trait_xp = {
                    trait = thuum_trait_track

			}
		}
	}

     type = character_event
	title = "Dismay"

        is_in_army = yes
		OR = {
			is_commanding_army = yes
		}
    }

	immediate = {
		save_scope_as = thuum_user
		if = {
			limit = {
				scope:thuum_user = {
					is_in_army = yes
					OR = {
						is_commanding_army = yes
					}
				}
			}
			if = {
				limit = {
					scope:thuum_user = {
						is_commanding_army = yes
					}

					location = {
						save_scope_as = sieging_location
					}
				}
			}

				knight_army = {
					location = {
						save_scope_as = sieging_location

                has_perk = dismay_perk
